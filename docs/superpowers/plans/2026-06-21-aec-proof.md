> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Answer the one make-or-break question for the whole Mac voice-loop *before* building the clean package: **does `speexdsp` AEC actually cancel the band's strong, co-located, BLE-delayed echo?** Capture a real echo recording from the band and run Speex AEC on it offline, measuring echo reduction (ERLE).

**Architecture:** Pure Mac-side, **no firmware change / no reflash** (offline proof aligns the two recordings by cross-correlation, so the `ts32` timestamp — needed only for *live* real-time alignment — is deferred to Plan B). Two steps: (1) extend `audio_tx.py --duplex` to *record* the reference (what we streamed down) and the mic (the uplink, decoded — which contains our speaker echo + the user's voice); (2) an offline tool that delay-aligns them, runs Speex `EchoCanceller`, and reports ERLE + a listenable cleaned WAV.

**Tech Stack:** Python 3, `bleak` (BLE), `liblc3` via ctypes (already used), `numpy` (cross-correlation/ERLE), `speexdsp` (Speex AEC, built from source). Device firmware is feature-complete and used as-is.

**Context for the implementer:** the band streams mic audio up only while in `MODE_DICTATION` (raise the band to your ear + speak). `tools/audio_tx.py` already streams a 16 kHz mono WAV *down* to the speaker and, with `--duplex`, subscribes to the uplink mic notify (`47A10002`, format `[seq16 LE][80-byte LC3 = 2×40B frames]`) and counts frames. `tools/audio_rx.py` already has a working `liblc3` *decoder* ctypes wrapper to copy from. Build/run notes are in `CLAUDE.md`.

```python
class Lc3Decoder:
    """Minimal ctypes wrapper over liblc3's decoder (mirror of Lc3Encoder)."""

    def __init__(self, lib_path):
        self.ok = False
        if not lib_path:
            return
        try:
            lib = ctypes.CDLL(lib_path)
            lib.lc3_decoder_size.restype = ctypes.c_uint
            lib.lc3_decoder_size.argtypes = [ctypes.c_int, ctypes.c_int]
            lib.lc3_setup_decoder.restype = ctypes.c_void_p
            lib.lc3_setup_decoder.argtypes = [
                ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_void_p]
            lib.lc3_decode.restype = ctypes.c_int
            lib.lc3_decode.argtypes = [
                ctypes.c_void_p, ctypes.c_void_p, ctypes.c_int,
                ctypes.c_int, ctypes.c_void_p, ctypes.c_int]
            size = lib.lc3_decoder_size(DT_US, SR_HZ)
            self._mem = ctypes.create_string_buffer(size)
            self._dec = lib.lc3_setup_decoder(DT_US, SR_HZ, 0, self._mem)
            self._lib = lib
            self._pcm = (ctypes.c_int16 * FRAME_SAMPLES)()
            self.ok = self._dec is not None
        except Exception as e:  # noqa: BLE001
            print(f"[decode disabled] {e}")
            self.ok = False

    def decode_frame(self, frame40: bytes) -> bytes:
        """40-byte LC3 frame -> 160 int16 samples (320 bytes)."""
        buf = (ctypes.c_uint8 * len(frame40)).from_buffer_copy(frame40)
        self._lib.lc3_decode(self._dec, buf, len(frame40), 0, self._pcm, 1)
        return bytes(self._pcm)
```

```python
#!/usr/bin/env python3
"""Synthetic unit test for the offline AEC (no hardware, no real recording).
Build reference + a known delayed/attenuated echo + a distinct near-end tone;
the AEC must cut the echo (high ERLE) while leaving the near-end mostly intact."""
import numpy as np
import aec_offline as aec

SR = 16000


def _tone(freq, n, amp=0.3):
    t = np.arange(n) / SR
    return (amp * np.sin(2 * np.pi * freq * t) * 32767).astype(np.int16)


def test_aec_cuts_synthetic_echo():
    n = SR * 4
    ref = _tone(800, n, 0.5)                       # "far" = what the speaker plays
    delay = 1200                                    # samples of echo path delay
    echo = np.zeros(n, dtype=np.int16)
    echo[delay:] = (ref[:n - delay] * 0.6).astype(np.int16)   # attenuated, delayed
    near = _tone(250, n, 0.25)                      # the user's voice (different freq)
    mic = np.clip(echo.astype(np.int32) + near.astype(np.int32), -32768, 32767).astype(np.int16)

    clean, est_delay, erle = aec.run_aec(ref, mic, sr=SR)

    assert abs(est_delay - delay) <= 256, f"delay est off: {est_delay} vs {delay}"
    assert erle >= 10.0, f"ERLE too low ({erle:.1f} dB) -- echo not cancelled"
    # near-end (250 Hz) energy should largely survive in the cleaned output
    near_energy = float(np.mean(near.astype(np.float64) ** 2))
    clean_energy = float(np.mean(clean.astype(np.float64) ** 2))
    assert clean_energy >= 0.2 * near_energy, "near-end voice was destroyed"


if __name__ == "__main__":
    for name, fn in sorted(globals().items()):
        if name.startswith("test_") and callable(fn):
            fn(); print(f"PASS {name}")
    print("all passed")
```

```python
#!/usr/bin/env python3
"""Offline acoustic echo cancellation proof for the gestureband voice-loop.

Loads <base>_ref.wav (reference = what the speaker played) and <base>_mic.wav
(mic = echo + the user's voice), estimates the bulk echo delay by cross-correlation,
aligns, runs Speex EchoCanceller, writes <base>_clean.wav, and prints ERLE (echo
return loss enhancement) -- the measure of how much echo was removed.

Usage:  python3 aec_offline.py echo1        # uses echo1_ref.wav + echo1_mic.wav
"""
import sys
import wave
import numpy as np

try:
    from speexdsp import EchoCanceller
except ImportError:
    EchoCanceller = None

SR = 16000
FRAME = 256              # Speex process() frame, samples
FILTER = 4096            # echo-tail filter length, samples (~256 ms residual after align)


def _read_wav(path):
    with wave.open(path, "rb") as w:
        assert w.getnchannels() == 1 and w.getsampwidth() == 2 and w.getframerate() == SR, \
            f"{path}: need 16 kHz mono 16-bit"
        return np.frombuffer(w.readframes(w.getnframes()), dtype=np.int16)


def _write_wav(path, pcm_i16):
    with wave.open(path, "wb") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(SR)
        w.writeframes(pcm_i16.astype(np.int16).tobytes())


def estimate_delay(ref, mic, max_delay=SR):
    """Bulk delay (samples) that mic lags ref, via FFT cross-correlation (fast --
    direct np.correlate would be O(n^2) and hang on multi-second clips). Returns the
    d in [0, max_delay] maximizing sum_t mic[t]*ref[t-d]."""
    n = min(len(ref), len(mic), SR * 8)
    r = ref[:n].astype(np.float64); m = mic[:n].astype(np.float64)
    r -= r.mean(); m -= m.mean()
    nfft = 1 << int(np.ceil(np.log2(2 * n)))
    corr = np.fft.irfft(np.fft.rfft(m, nfft) * np.conj(np.fft.rfft(r, nfft)), nfft)
    return int(np.argmax(corr[:max_delay + 1]))


def erle_db(echo_ref, residual):
    """ERLE = 10 log10( power(mic-echo) / power(residual) ), over the aligned region."""
    e = float(np.mean(echo_ref.astype(np.float64) ** 2)) + 1e-9
    r = float(np.mean(residual.astype(np.float64) ** 2)) + 1e-9
    return 10.0 * np.log10(e / r)


def run_aec(ref, mic, sr=SR):
    """Returns (clean_i16, est_delay, erle_db). Pure function -> unit-testable."""
    if EchoCanceller is None:
        raise RuntimeError("speexdsp not installed (see plan Task 2 Step 1)")
    delay = estimate_delay(ref, mic)
    # Pre-align: delay the reference by `delay` so it lines up with the mic.
    far = np.concatenate([np.zeros(delay, dtype=np.int16), ref])[:len(mic)]
    if len(far) < len(mic):
        far = np.concatenate([far, np.zeros(len(mic) - len(far), dtype=np.int16)])
    ec = EchoCanceller.create(FRAME, FILTER, sr)
    out = bytearray()
    for i in range(0, len(mic) - FRAME + 1, FRAME):
        near_b = mic[i:i + FRAME].tobytes()
        far_b = far[i:i + FRAME].tobytes()
        out.extend(ec.process(near_b, far_b))
    clean = np.frombuffer(bytes(out), dtype=np.int16)
    nn = len(clean)
    erle = erle_db(mic[:nn], clean)
    return clean, delay, erle


def main():
    if len(sys.argv) != 2:
        sys.exit("usage: python3 aec_offline.py <basename>  (uses <base>_ref.wav + <base>_mic.wav)")
    base = sys.argv[1]
    ref = _read_wav(f"{base}_ref.wav")
    mic = _read_wav(f"{base}_mic.wav")
    clean, delay, erle = run_aec(ref, mic)
    _write_wav(f"{base}_clean.wav", clean)
    print(f"delay estimate: {delay} samples ({delay/SR*1000:.0f} ms)")
    print(f"ERLE: {erle:.1f} dB   (higher = more echo removed; >~12 dB is good)")
    print(f"wrote {base}_clean.wav -- listen: echo should be down, your voice should remain")


if __name__ == "__main__":
    main()
```
